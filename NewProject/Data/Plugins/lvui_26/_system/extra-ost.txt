栞 playbgm _name
{
    臨時全域変数 play_status = 2
    臨時全域変数 ost_playing_y = 0
    
    もし (_name == bgm01) {
        臨時全域変数 bgm_playing = 1
        臨時全域変数 ost_playing_y = 111
    }
    違ってももし (_name == bgm02) {
        臨時全域変数 bgm_playing = 2
        臨時全域変数 ost_playing_y = 183
    }
    違ってももし (_name == bgm03) {
        臨時全域変数 bgm_playing = 3
        臨時全域変数 ost_playing_y = 254
    }
    違ってももし (_name == bgm04) {
        臨時全域変数 bgm_playing = 4
        臨時全域変数 ost_playing_y = 326
    }
    違ってももし (_name == bgm05) {
        臨時全域変数 bgm_playing = 5
        臨時全域変数 ost_playing_y = 397
    }
    
    背景音 {{_name}}.ogg ost_volume 反復
    スクリプト 現在 update_playing_anim
}
スクリプト終了
