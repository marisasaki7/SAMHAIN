栞 load_after
{

    // After load, rendering is disabled by default — re-enable it.
    描画許可 true

    待機 300

    アウト sys.* 全画面領域

    // Resume all paused effects (across all screenspaces).
    再生続行 "(?!vn_sysBGM).*" 全画面領域

    アウト event_black 300 全画面領域

    スクリプト /Plugins/lvui_26/_system/textbox.txt txt_show

    // Restore normal text advance mode.
    文字進行通常強制 false

    // After load, input is disabled by default — re-enable it.
    入力禁止 false
